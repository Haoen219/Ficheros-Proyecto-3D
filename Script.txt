using UnityEngine;

public class BreadCatMovement : MonoBehaviour
{
    public float rotationSpeed = 100f; // Velocidad de rotación
    public float moveSpeed = 2f;       // Velocidad de movimiento vertical
    public float moveAmplitude = 0.5f; // Amplitud del movimiento vertical

    private float initialY;
    private bool isMoving = false;
    private bool isRotating = false;

    void Start()
    {
        initialY = transform.position.y;
    }

    void OnMouseDown()
    {
        ToggleRotation();
        ToggleMovement();
    }

    void ToggleRotation()
    {
        isRotating = !isRotating;
        if (isRotating)
            InvokeRepeating(nameof(RotateObject), 0f, Time.deltaTime);
        else
            CancelInvoke(nameof(RotateObject));
    }

    void ToggleMovement()
    {
        isMoving = !isMoving;
        if (isMoving)
            InvokeRepeating(nameof(MoveObject), 0f, Time.deltaTime);
        else
            CancelInvoke(nameof(MoveObject));
    }

    void RotateObject()
    {
        transform.Rotate(0, 0, rotationSpeed * Time.deltaTime);
    }

    void MoveObject()
    {
        float newY = initialY + Mathf.Sin(Time.time * moveSpeed) * moveAmplitude;
        transform.position = new Vector3(transform.position.x, newY, transform.position.z);
    }
}
